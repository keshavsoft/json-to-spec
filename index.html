Usage

import { compile } from "./src/index.js";

// 1. structure.json (UI blueprint)
// 2. data.json (data collections & values)
const compiledSpec = compile({
    inStructure: structure,
    inData: data
});

// Pass compiledSpec to json-to-dom to build live elements
